add_library(mterm_core STATIC
  core/Policy.cpp core/FileService.cpp core/Canvas.cpp core/Store.cpp)
target_include_directories(mterm_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mterm_core PUBLIC Qt6::Core Qt6::Sql)
if(MSVC)
  target_compile_options(mterm_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(mterm_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(mterm-core-tests tests/CoreTests.cpp)
  target_link_libraries(mterm-core-tests PRIVATE mterm_core Qt6::Test)
  add_test(NAME native-core COMMAND mterm-core-tests -o core-tests.txt,txt)
endif()

add_library(mterm_runtime STATIC runtime/ProcessRunner.cpp runtime/ProcessInventory.cpp runtime/JsonLines.cpp)
target_include_directories(mterm_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mterm_runtime PUBLIC mterm_core Qt6::Core)
if(WIN32)
 target_link_libraries(mterm_runtime PRIVATE psapi)
endif()
if(BUILD_TESTING)
 add_executable(mterm-test-child tests/TestChild.cpp)
 target_link_libraries(mterm-test-child PRIVATE Qt6::Core)
 add_executable(mterm-runtime-tests tests/RuntimeTests.cpp)
 target_link_libraries(mterm-runtime-tests PRIVATE mterm_runtime Qt6::Test)
 add_dependencies(mterm-runtime-tests mterm-test-child)
 add_test(NAME native-runtime COMMAND mterm-runtime-tests -o runtime-tests.txt,txt)
 set_tests_properties(native-core PROPERTIES COMMAND_EXPAND_LISTS TRUE)
endif()

add_library(mterm_services STATIC services/Backend.cpp services/WorkspaceSession.cpp services/JobService.cpp)
target_link_libraries(mterm_services PUBLIC mterm_runtime)
if(BUILD_TESTING)
 add_executable(mterm-backend-tests tests/BackendTests.cpp)
 target_link_libraries(mterm-backend-tests PRIVATE mterm_services Qt6::Test)
 add_dependencies(mterm-backend-tests mterm-test-child)
 add_test(NAME native-backend COMMAND mterm-backend-tests -o backend-tests.txt,txt)
endif()
